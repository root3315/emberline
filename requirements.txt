# Python side of Emberline: the model, the weather, the forecast.
#
# The site itself needs none of this — `dist/` is static and the data files
# it reads are committed. This is for rebuilding those files.
#
#     python -m venv .venv
#     .venv\Scripts\activate          # Windows
#     pip install -r requirements.txt
#
# Versions are the ones the shipped model and forecast were built with.
# scikit-learn in particular: a pickled HistGradientBoostingClassifier is
# not guaranteed to load across minor versions, and `models/fire-7d.pkl`
# was written by the one below.

# ── always needed ──────────────────────────────────────────────────────
numpy==2.2.6
scikit-learn==1.7.2
scipy==1.15.3           # pulled in by sklearn; pinned so it does not drift
joblib==1.5.3           # ditto

# ── only for fetching weather, and only once ───────────────────────────
# `fetch_arco.py` reads the public ERA5 mirror over HTTP and unpacks Zarr
# chunks itself, which is why numcodecs is here and zarr is not.
numcodecs==0.13.1

# `fetch_gfs.py` reads GRIB2 from NOAA for the live forecast week.
eccodes==2.48.0

# `fetch_era5.py` is the alternative route through Copernicus, and needs a
# CDS_KEY in .env. Not required if you use fetch_arco.py.
cdsapi==0.7.7

# ── optional, and it did not earn its place ────────────────────────────
# `fetch_population.py` reads the GHSL settlement raster. The features it
# builds were measured and dropped: +0.0001 average precision, ranked 12th
# and 13th of 27. Kept because the negative result is part of the record.
rasterio==1.4.4
